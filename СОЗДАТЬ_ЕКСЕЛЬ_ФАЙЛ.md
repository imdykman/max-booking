    == пример ==  
const XLSX = require('xlsx');

const wb = XLSX.utils.book_new();

// Общие
const ws1 = XLSX.utils.aoa_to_sheet([
    ['Параметр', 'Значение'],
    ['Название_техникума', ''],
    ['Адрес', ''],
    ['Телефон', ''],
]);
XLSX.utils.book_append_sheet(wb, ws1, 'Общие');

// Бюджет
const ws2 = XLSX.utils.aoa_to_sheet([
    ['Название', 'Срок', 'Квалификация'],
    ['', '', ''],
]);
XLSX.utils.book_append_sheet(wb, ws2, 'Бюджет');

// Документы
const ws3 = XLSX.utils.aoa_to_sheet([
    ['№', 'Документ'],
    [1, 'Паспорт'],
    [2, 'Аттестат'],
]);
XLSX.utils.book_append_sheet(wb, ws3, 'Документы');

XLSX.writeFile(wb, 'ШАБЛОН_ТЕХНИКУМА.xlsx');
console.log('? Готово');
== конец примера ==

5. Запустить
cmd
node create_template.js
6. Готово
Появится ШАБЛОН_ТЕХНИКУМА.xlsx — отправляйте техникуму.
